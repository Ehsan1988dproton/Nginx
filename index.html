async function syncWorkerSubscription() { try { // ۱. تنظیم کادر کپی و کیوآرکد روی لینک سابسکریپشن const subInput = document.getElementById('sub-url-input'); if (subInput) subInput.value = WORKER_SUB_URL; if (typeof generateQrCode === 'function') { generateQrCode(WORKER_SUB_URL); } // ۲. ارسال درخواست به ورکر const response = await fetch(WORKER_SUB_URL); // ۳. خواندن هدر اطلاعات ترافیک (Subscription-Userinfo) const userInfo = response.headers.get('subscription-userinfo'); if (userInfo) { const stats = {}; userInfo.split(';').forEach(p => { const [k, v] = p.trim().split('='); if (k && v) stats[k] = parseInt(v, 10); }); const upload = stats.upload || 0; const download = stats.download || 0; const total = stats.total || (10 * 1024 * 1024 * 1024); // ۱۰ گیگابایت پیش‌فرض در صورت نبود هدر const used = upload + download; const remaining = Math.max(0, total - used); const toGB = b => (b / (1024 ** 3)).toFixed(2); const toMB = b => (b / (1024 ** 2)).toFixed(2); const elRemaining = document.querySelector('#stat-remaining'); if (elRemaining) elRemaining.innerText = toGB(remaining) + ' GB'; const elTotal = document.querySelector('#stat-total'); if (elTotal) elTotal.innerText = 'of ' + toGB(total) + ' GB'; const elDown = document.querySelector('#stat-download'); if (elDown) elDown.innerText = toMB(download) + ' MB'; const elUp = document.querySelector('#stat-upload'); if (elUp) elUp.innerText = toMB(upload) + ' MB'; const percentage = total > 0 ? Math.min(100, (used / total) * 100).toFixed(1) : 0; const elPercent = document.querySelector('#stat-percentage'); if (elPercent) elPercent.innerText = percentage + '%'; if (stats.expire) { const expDate = new Date(stats.expire * 1000); const elExp = document.querySelector('#stat-expiry'); if (elExp) elExp.innerText = expDate.toISOString().split('T')[0]; const days = Math.max(0, Math.ceil((expDate - new Date()) / (1000 * 60 * 60 * 24))); const elDays = document.querySelector('#stat-days-left'); if (elDays) elDays.innerText = days; } } // ۴. دیکود کردن کانفیگ‌های داخل ساب (Base64) const rawBody = await response.text(); let decoded = rawBody; try { decoded = atob(rawBody.trim()); } catch (e) { // در صورتی که پاسخ پلین‌تکست باشد } const configs = decoded.split('\n').filter(c => c.trim().startsWith('vless://') || c.trim().startsWith('vmess://') || c.trim().startsWith('trojan://')); // ۵. درج کانفیگ‌ها در تب Configs if (typeof renderConfigsList === 'function') { renderConfigsList(configs); } } catch (err) { console.error("خطا در همگام‌سازی با ورکر:", err); } } // فراخوانی هنگام لود کامل صفحه window.addEventListener('DOMContentLoaded', syncWorkerSubscription);
